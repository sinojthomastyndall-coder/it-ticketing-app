function exportToExcel() { if (tickets.length === 0) { alert('No tickets available to export.'); return; } const headers = ["Ticket ID", "End User", "Subject", "Sub Details", "Location", "Description", "Status", "Priority", "Assignee"]; const rows = tickets.map(t => { let subDetail = t.subOption || ''; if(t.deskItems && t.deskItems.length > 0) { subDetail = t.deskItems.map(i => `${i.name}: ${i.qty}`).join('; '); } let locationStr = 'N/A'; if (t.locBlockFloor) { const blockCode = t.locBlockFloor.split(' ')[0]; const secTag = (t.locSection && blockCode.startsWith('A')) ? ` (${t.locSection})` : ''; locationStr = `${blockCode}${secTag}, ${t.locRoom}, ${t.locSeat}`; } else if (t.location) { locationStr = t.location; } const cleanDesc = (t.desc || '').replace(/\r?\n|\r/g, ' '); return [ `"${t.id}"`, `"${(t.endUser || '').replace(/"/g, '""')}"`, `"${(t.subject || '').replace(/"/g, '""')}"`, `"${subDetail.replace(/"/g, '""')}"`, `"${locationStr.replace(/"/g, '""')}"`, `"${cleanDesc.replace(/"/g, '""')}"`, `"${t.status}"`, `"${t.priority}"`, `"${(t.assignee || '').replace(/"/g, '""')}"` ]; }); const csvContent = "\uFEFF" + [headers.join(','), ...rows.map(e => e.join(','))].join('\n'); // Convert CSV content to base64 encoding for reliable WebView file handling const encodedUri = "data:text/csv;charset=utf-8;base64," + btoa(unescape(encodeURIComponent(csvContent))); const link = document.createElement('a'); link.setAttribute('href', encodedUri); link.setAttribute('download', `IT_Tickets_Export_${new Date().toISOString().slice(0,10)}.csv`); // Open in a new window/tab as fallback for Android WebView download interceptors link.setAttribute('target', '_blank'); document.body.appendChild(link); link.click(); document.body.removeChild(link); }
